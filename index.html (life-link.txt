<script>
  // Alerts are stored in the browser (localStorage) under this key.
  // This is why it works without a backend: it is a prototype.
  var STORAGE_KEY = "lifelink_alerts";

  // Location chosen by the user (stays null unless they allow it)
  var coords = null;

  // ---------- Helpers to read and save alerts ----------
  function loadAlerts() {
    try {
      return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
    } catch (e) {
      return [];
    }
  }
  function saveAlerts(list) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
  }

  // ---------- Switch between the two screens ----------
  function showScreen(which) {
    document.getElementById("screenUser").classList.toggle("active", which === "user");
    document.getElementById("screenDash").classList.toggle("active", which === "dash");
    document.getElementById("tabUser").classList.toggle("active", which === "user");
    document.getElementById("tabDash").classList.toggle("active", which === "dash");
    if (which === "dash") renderAlerts();
    window.scrollTo(0, 0);
  }

  // ---------- User screen ----------
  function openForm() {
    document.getElementById("sosForm").classList.remove("hidden");
    document.getElementById("sentBox").classList.add("hidden");
    document.getElementById("name").focus();
    document.getElementById("sosForm").scrollIntoView({ behavior: "smooth" });
  }

  // Ask the browser for the user's location (the browser shows a permission popup)
  function getLocation() {
    var status = document.getElementById("locStatus");
    status.className = "";
    if (!navigator.geolocation) {
      status.className = "bad";
      status.textContent = "This browser cannot share location. Please type where you are.";
      return;
    }
    status.textContent = "Waiting for permission...";
    navigator.geolocation.getCurrentPosition(
      function (pos) {
        coords = { lat: pos.coords.latitude, lng: pos.coords.longitude };
        status.className = "ok";
        status.textContent = "Location added (" + coords.lat.toFixed(4) + ", " + coords.lng.toFixed(4) + ")";
      },
      function () {
        coords = null;
        status.className = "bad";
        status.textContent = "Location not shared. Please type where you are below.";
      },
      { enableHighAccuracy: true, timeout: 10000 }
    );
  }

  // Create the alert and save it
  function sendAlert(event) {
    event.preventDefault();

    var manual = document.getElementById("manualLoc").value.trim();
    if (!coords && !manual) {
      var status = document.getElementById("locStatus");
      status.className = "bad";
      status.textContent = "Please share your location or type where you are.";
      return;
    }

    var alerts = loadAlerts();
    alerts.unshift({
      id: Date.now(),
      createdAt: new Date().toISOString(),
      name: document.getElementById("name").value.trim(),
      phone: document.getElementById("phone").value.trim(),
      type: document.getElementById("type").value,
      blood: document.getElementById("blood").value,
      contact: document.getElementById("contact").value.trim(),
      medical: document.getElementById("medical").value.trim(),
      coords: coords,
      manualLocation: manual,
      status: "New",
      acknowledgedAt: null
    });
    saveAlerts(alerts);

    document.getElementById("sosForm").classList.add("hidden");
    document.getElementById("sentBox").classList.remove("hidden");
    document.getElementById("sentBox").scrollIntoView({ behavior: "smooth" });
  }

  function resetUser() {
    document.getElementById("sosForm").reset();
    document.getElementById("locStatus").className = "";
    document.getElementById("locStatus").textContent = "Location is only used if you allow it.";
    coords = null;
    document.getElementById("sentBox").classList.add("hidden");
    window.scrollTo(0, 0);
  }

  // ---------- Responder dashboard ----------
  function formatTime(iso) {
    return new Date(iso).toLocaleString([], { day: "numeric", month: "short", hour: "2-digit", minute: "2-digit" });
  }

  // Add one row (label + value) to an alert card, only if there is a value
  function addRow(dl, label, value) {
    if (!value) return;
    var dt = document.createElement("dt");
    dt.textContent = label;
    var dd = document.createElement("dd");
    dd.textContent = value;
    dl.appendChild(dt);
    dl.appendChild(dd);
  }

  function renderAlerts() {
    var list = loadAlerts();
    var box = document.getElementById("alertList");
    box.innerHTML = "";

    var newCount = list.filter(function (a) { return a.status === "New"; }).length;
    document.getElementById("counts").textContent =
      list.length + " alert(s), " + newCount + " waiting to be acknowledged";

    if (list.length === 0) {
      var empty = document.createElement("p");
      empty.className = "empty";
      empty.textContent = "No alerts yet. Create one from the I need help tab.";
      box.appendChild(empty);
      return;
    }

    list.forEach(function (a) {
      var done = a.status === "Acknowledged";
      var card = document.createElement("article");
      card.className = "alert-card" + (done ? " done" : "");

      var top = document.createElement("div");
      top.className = "alert-top";
      var h = document.createElement("h3");
      h.textContent = a.type + " - " + a.name;
      var badge = document.createElement("span");
      badge.className = "badge" + (done ? " done" : "");
      badge.textContent = done ? "Acknowledged" : "New";
      top.appendChild(h);
      top.appendChild(badge);
      card.appendChild(top);

      var dl = document.createElement("dl");
      addRow(dl, "Received", formatTime(a.createdAt));
      addRow(dl, "Phone", a.phone);
      addRow(dl, "Blood group", a.blood);
      addRow(dl, "Medical notes", a.medical);
      addRow(dl, "Emergency contact", a.contact);
      if (a.coords) addRow(dl, "Location", a.coords.lat.toFixed(5) + ", " + a.coords.lng.toFixed(5));
      addRow(dl, "Described place", a.manualLocation);
      if (done) addRow(dl, "Acknowledged", formatTime(a.acknowledgedAt));
      card.appendChild(dl);

      var actions = document.createElement("div");
      actions.className = "alert-actions";

      if (a.coords) {
        var map = document.createElement("a");
        map.className = "btn btn-light";
        map.textContent = "Open in maps";
        map.target = "_blank";
        map.rel = "noopener";
        map.href = "https://www.google.com/maps?q=" + a.coords.lat + "," + a.coords.lng;
        actions.appendChild(map);
      }
      var call = document.createElement("a");
      call.className = "btn btn-light";
      call.textContent = "Call " + a.name.split(" ")[0];
      call.href = "tel:" + a.phone.replace(/[^0-9+]/g, "");
      actions.appendChild(call);

      if (!done) {
        var ack = document.createElement("button");
        ack.className = "btn btn-green";
        ack.textContent = "Acknowledge";
        ack.onclick = function () { acknowledge(a.id); };
        actions.appendChild(ack);
      }
      card.appendChild(actions);
      box.appendChild(card);
    });
  }

  function acknowledge(id) {
    var list = loadAlerts();
    list.forEach(function (a) {
      if (a.id === id) {
        a.status = "Acknowledged";
        a.acknowledgedAt = new Date().toISOString();
      }
    });
    saveAlerts(list);
    renderAlerts();
  }

  function clearAlerts() {
    if (confirm("Delete all demo alerts?")) {
      saveAlerts([]);
      renderAlerts();
    }
  }

  // If another tab of this browser creates or acknowledges an alert, update this one.
  // (This is how you can demo two screens side by side. Real cross-device sending needs a backend.)
  window.addEventListener("storage", function (e) {
    if (e.key === STORAGE_KEY) renderAlerts();
  });

  renderAlerts();
</script>
